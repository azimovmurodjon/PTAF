# ==================================
# REPORTING CONFIGURATION
# ==================================
#
# Controls the generation of per-feature-file Extent Reports.
# By default, PTAF generates one combined Extent HTML report for all scenarios in a run.
# When per_feature_reports_enabled is set to true, PTAF additionally generates one
# individual Extent HTML report per feature file — each titled with the Feature Name
# declared in the feature file (e.g. "Feature: User Authentication").
#
# This works for ALL automation types: UI, Mobile, API, Database, Performance, PDF, XML, CSV.
# The combined report is always generated regardless of this setting.
#
reporting:
  # Set to true to generate one individual Extent HTML report per feature file.
  # Each report is titled with the Feature Name from the feature file's "Feature:" declaration.
  # Set to false (default) to generate only the combined report — no change to current behavior.
  per_feature_reports_enabled: true

  # Output directory for per-feature HTML reports.
  # Each report is saved as: {output_dir}/{feature_file_name}_{timestamp}.html
  # The directory is created automatically if it does not exist.
  per_feature_reports_output_dir: "test-output/per-feature-reports"

  # Set to true to also generate a PDF version of each per-feature report.
  # Requires per_feature_reports_enabled: true to have any effect.
  # Set to false (default) to generate HTML reports only.
  per_feature_pdf_enabled: false

  # Set to true to also generate a Glass-style PDF (via cucumber-pdf-report library) for each
  # per-feature report. These are written to a SEPARATE directory (per_feature_glass_pdf_output_dir)
  # so they do not mix with the PTAF direct PDFs from per_feature_pdf_enabled.
  # Requires per_feature_reports_enabled: true to have any effect.
  # Note: Glass PDF generation spawns one subprocess per feature (adds ~1-2 seconds per feature).
  per_feature_glass_pdf_enabled: true

  # Output directory for per-feature Glass-style PDF reports.
  # Each report is saved as: {output_dir}/{feature_file_name}_{timestamp}.pdf
  # The directory is created automatically if it does not exist.
  per_feature_glass_pdf_output_dir: "test-output/per-feature-reports-glass"


# ==================================
# SOFT ASSERTIONS CONFIGURATION
# ==================================
#
# Controls whether the framework uses soft assertions (continue-on-failure) mode.
#
# When enabled: false (default)
#   - Normal automation behavior. A step fails → screenshot captured → test stops → browser closes.
#   - This is the recommended mode for most test runs.
#
# When enabled: true
#   - Soft assertion mode. A step fails → framework retries the step for retry_seconds →
#     if still failing, screenshot is captured → execution continues to the next step →
#     at the end of the scenario, if any steps failed, the scenario is marked as FAILED
#     with a full summary of all failed steps.
#   - The browser is kept open throughout the scenario so all steps can run.
#   - Works for: UI automation, mobile browser simulation, mobile native app, native mobile browser.
#   - Session crashes (driver disconnected, app crashed) still stop the test immediately.
#
soft_assertions:
  # Set to true to enable soft assertion mode.
  # Set to false (default) for normal fail-fast behavior — no change to current behavior.
  enabled: false

  # How many seconds to retry a failed step before giving up and moving on to the next step.
  # Only used when enabled: true.
  # Recommended range: 1 to 10 seconds. Default: 3 seconds.
  # Keep this low — if an element has not appeared in 3 seconds it is almost certainly a real failure.
  retry_seconds: 3
