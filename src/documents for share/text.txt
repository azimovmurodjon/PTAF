@When("we click all available documents on doc frame")
public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
    int total = ReturnTotalNumberOfElement();
    String basePath = ConfigurationProperties.getValue("downloadDocument");
    String ext = ".pdf";
    String screenshotPath = "test-output/screenshots/body.png";

    if (total <= 0) {
        weCaptureScreenshotOnDirectorPage("director_page", "body", screenshotPath);
        weReportStringValue("[INFO]", "No documents available to download.");
        return;
    }

    for (int i = 1; i <= total; i++) {
        String key = toDocumentKey(i);
        String targetFile = String.format("%s_%02d%s", basePath, i, ext);

        try {
            // 1) Open/preview document
            weDoubleClickActionOnDirectorDocFrame("director_page", key);
            Thread.sleep(300); // small settle (replace with your wait util if you have)

            // 2) Decide what to do based on visibility
            boolean isDepositTitleVisible = weVerifyOnDirectorMarkupFrameIsBooleanVisible(
                    "director_page",
                    "deposit_main_title"
            );

            if (isDepositTitleVisible) {
                // Visible => ONLY screenshot, DO NOT download, DO NOT run else
                weCaptureScreenshotOnDirectorPage("director_page", "body", screenshotPath);
                System.out.printf("[INFO] deposit_main_title is visible. Skipping download for key=%s%n", key);

            } else {
                // Not visible => try download
                try {
                    System.out.println("[INFO] Try to Download");
                    weDownloadOnDirectorDocFrame("director_page", key, targetFile);
                    weCaptureScreenshotOnDirectorPage("director_page", "body", screenshotPath);
                    System.out.printf("[INFO] Downloaded %s using key %s%n", targetFile, key);
                } catch (Exception e) {
                    System.out.printf("[WARN] Failed to download %s (%s): %s%n",
                            targetFile, key, e.getMessage());
                }
            }

        } catch (Exception e) {
            System.out.printf("[WARN] Failed while opening/processing doc key=%s: %s%n", key, e.getMessage());

        } finally {
            // 3) IMPORTANT: always go back / close preview so next loop starts clean
            try {
                // Use whichever you already have in framework:
                // - press ESC
                // - click Close/Back button
                // - page.goBack()
                // Examples (pick ONE that matches your app):
                // wePressKeyOnDirectorPage("director_page", "Escape");
                // weClickActionOnDirectorMarkupFrame("director_page", "close_preview_btn");
                // weGoBackOnDirectorPage("director_page");

                wePressKeyOnDirectorPage("director_page", "Escape"); // <-- recommended if your preview closes on ESC
                Thread.sleep(200);
            } catch (Exception ignore) {
                // don’t fail the loop if close/back fails; continue to next document
            }
        }

        Thread.sleep(300);
    }
}
