@When("we click all available documents on doc frame")
public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
    int total = ReturnTotalNumberOfElement();
    String basePath = ConfigurationProperties.getValue("downloadDocument");
    String ext = ".pdf";
    String filePath = "test-output/screenshots/body.png";

    if (total <= 0) {
        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        weReportStringValue("[INFO]", "No documents available to download.");
        return;
    }

    for (int i = 1; i <= total; i++) {
        String key = toDocumentKey(i);
        String targetFile = String.format("%s_%02d%s", basePath, i, ext);

        try {
            // 1) Open / preview document
            weDoubleClickActionOnDirectorDocFrame("director_page", key);

            // optional small pause if your app needs it; ideally replace with proper waits
            Thread.sleep(800);

            // 2) Check marker element (TRUE if visible, FALSE if not visible)
            boolean isVisible = isDepositTitleVisibleSafe("director_page", "deposit_main_title");

            if (isVisible) {
                weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                System.out.printf("[INFO] deposit_main_title visible => screenshot only, skip download (key=%s)%n", key);
                continue; // go next document
            }

            // 3) If NOT visible => download
            System.out.printf("[INFO] deposit_main_title NOT visible => download (key=%s)%n", key);
            downloadWithScreenshot(key, targetFile, filePath);

        } catch (Exception e) {
            // 4) Fallback: if open/check fails, still TRY download
            System.out.printf("[WARN] Failed to open/check key=%s (%s). Will try download anyway.%n", key, e.getMessage());
            downloadWithScreenshot(key, targetFile, filePath);
        }
    }
}

/**
 * IMPORTANT: This returns TRUE only if element is visible.
 * If your uiAssert throws when not visible, we catch and return FALSE.
 */
private boolean isDepositTitleVisibleSafe(String element, String locator) {
    try {
        uiAssert.isVisible(
                director_page,
                director_doc_frame,
                director_doc_frameViewer,
                director_doc_frame_markup_frame,
                element,
                locator
        );
        return true;
    } catch (Exception ignored) {
        return false;
    }
}

/**
 * Download helper (keeps your existing screenshot + logging pattern).
 */
private void downloadWithScreenshot(String key, String targetFile, String filePath) {
    try {
        weDownloadOnDirectorDocFrame("director_page", key, targetFile);
        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        System.out.printf("[INFO] Downloaded %s using key %s%n", targetFile, key);
        Thread.sleep(500);
    } catch (Exception ex) {
        System.out.printf("[WARN] Failed to download %s (key=%s): %s%n", targetFile, key, ex.getMessage());
        // If you want the whole test to STOP on download failure, throw here:
        // throw new RuntimeException("Download failed for key=" + key + ": " + ex.getMessage(), ex);
    }
}
