@When("we click all available documents on doc frame")
public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
    int total = ReturnTotalNumberOfElement();
    String basePath = ConfigurationProperties.getValue("downloadDocument");
    String ext = ".pdf";
    String filePath = "test-output/screenshots/body.png";

    if (total <= 0) {
        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        weReportStringValue("[INFO]", "No documents available to download.");
        return;
    }

    for (int i = 1; i <= total; i++) {
        String key = toDocumentKey(i);
        String targetFile = String.format("%s_%02d%s", basePath, i, ext);

        try {
            // 1) Open the document first
            weDoubleClickActionOnDirectorDocFrame("director_page", key);
            Thread.sleep(800);

            // 2) TRY DOWNLOAD FIRST (most important)
            boolean downloaded = tryDownloadWithScreenshot(key, targetFile, filePath);
            if (downloaded) {
                System.out.printf("[INFO] Downloaded successfully (key=%s)%n", key);
                continue;
            }

            // 3) If not downloadable => screenshot-only logic
            boolean isVisible = isDepositTitleVisibleSafe("director_page", "deposit_main_title");
            if (isVisible) {
                weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                System.out.printf("[INFO] Not downloadable; deposit_main_title visible => screenshot only (key=%s)%n", key);
            } else {
                // If neither download worked nor title visible, still capture evidence
                weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                System.out.printf("[WARN] Not downloaded and title not visible => screenshot for evidence (key=%s)%n", key);
            }

        } catch (Exception e) {
            // If anything fails, still attempt download, then screenshot
            System.out.printf("[WARN] Failed to open/check key=%s (%s). Trying download anyway...%n", key, e.getMessage());

            boolean downloaded = tryDownloadWithScreenshot(key, targetFile, filePath);
            if (!downloaded) {
                weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                System.out.printf("[WARN] Fallback: download failed too => screenshot only (key=%s)%n", key);
            }
        }
    }
}

/**
 * Returns TRUE only if download succeeded.
 * If element/download is not available it returns FALSE (no exception bubbles up).
 */
private boolean tryDownloadWithScreenshot(String key, String targetFile, String filePath) {
    try {
        weDownloadOnDirectorDocFrame("director_page", key, targetFile);
        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        Thread.sleep(500);
        return true;
    } catch (Exception ex) {
        System.out.printf("[INFO] Download not available or failed for key=%s: %s%n", key, ex.getMessage());
        return false;
    }
}

/**
 * Returns TRUE only if visible; FALSE if not visible (no throw).
 */
private boolean isDepositTitleVisibleSafe(String element, String locator) {
    try {
        uiAssert.isVisible(
                director_page,
                director_doc_frame,
                director_doc_frameViewer,
                director_doc_frame_markup_frame,
                element,
                locator
        );
        return true;
    } catch (Exception ignored) {
        return false;
    }
}
