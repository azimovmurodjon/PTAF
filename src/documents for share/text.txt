@When("we click all available documents on doc frame")
public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
    int total = ReturnTotalNumberOfElement();
    String basePath = ConfigurationProperties.getValue("downloadDocument"); // e.g. C:/Downloads/document
    String ext = ".pdf";
    String filePath = "test-output/screenshots/body.png";

    if (total <= 0) {
        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        weReportStringValue("[INFO]", "No documents available to download.");
        return;
    }

    for (int i = 1; i <= total; i++) {
        String key = toDocumentKey(i);
        String targetFile = String.format("%s_%02d%s", basePath, i, ext);

        // --- STEP 1: open doc + screenshot (this part you said works perfectly) ---
        try {
            Thread.sleep(800); // keep small pause
            weDoubleClickActionOnDirectorDocFrame("director_page", key);
            Thread.sleep(400); // allow viewer to render
            weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
        } catch (Throwable t) {
            System.out.printf("[WARN] Failed to open/screenshot doc #%d (key=%s): %s%n", i, key, t.getMessage());
            // even if open fails, still try download attempt (some apps allow download without viewer)
        }

        // --- STEP 2: try download (DO NOT STOP if it fails) ---
        try {
            weDownloadOnDirectorDocFrame("director_page", key, targetFile);
            System.out.printf("[INFO] Downloaded %s using key %s%n", targetFile, key);
        } catch (Throwable t) {
            // This must NOT stop the loop (doc may be not downloadable)
            System.out.printf("[INFO] Not downloadable or download failed for doc #%d (key=%s): %s%n", i, key, t.getMessage());
        }

        // --- STEP 3: small pause between documents ---
        Thread.sleep(300);
    }
}
