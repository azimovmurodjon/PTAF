package com.ptaf.ui.stepdefinitions;

import com.microsoft.playwright.BrowserContext;
import com.microsoft.playwright.Page;
import com.microsoft.playwright.options.AriaRole;
import com.ptaf.hooks.Hooks;
import com.ptaf.ui.helpers.TestContext;
import com.ptaf.ui.pages.FrameCommonMethods;
import com.ptaf.ui.pages.PageCommonMethods;
import com.ptaf.utils.ConfigurationProperties;
import io.cucumber.java.en.And;
import io.cucumber.java.en.Then;
import io.cucumber.java.en.When;

public class NewPageCommonSteps {

    // Base page/context from Hooks
    private final Page page = Hooks.getPage();
    private final BrowserContext context = Hooks.getContext();

    private static final String filePath = ConfigurationProperties.getExcelDocumentLocation();
    public static String trackindID = null;
    public static String dtrackindID = null;
    public static String dashtrackindID = null;

    private final TestContext testContext;

    // Popup page + helpers – initialized lazily after popup opens
    private Page new_page;
    private PageCommonMethods pageCommonMethods;
    private FrameCommonMethods frameCommonMethods;

    // Frames (kept as in your original code)
    //    private static final String plad_frame = "iframe[title=\"Plaid Link\"]";
    //    private static final String plad_frame ="#AcceptUIContainer iframe";
    private static final String plad_frame = "#plaid-link-iframe-2";
    private static final String plad_frame_B = "#plaid-link-iframe-1";
    private static final String pop_frame = "//*[@id=\"AcceptUIContainer\"]/iframe";
    //    private static final String pop_frame = "iframe";
    private static final String atomic_frame = "#atomic-transact-iframe";

    public NewPageCommonSteps(TestContext testContext) {
        this.testContext = testContext;
    }

    /**
     * Utility method used by other steps to make sure the popup page
     * and helper classes are initialized before use.
     */
    private void ensurePopupInitialized() {
        if (new_page == null) {
            throw new IllegalStateException(
                    "Popup page (new_page) is not initialized. " +
                    "Make sure you call the step that opens the new page (popup) first."
            );
        }
        if (pageCommonMethods == null) {
            pageCommonMethods = new PageCommonMethods(new_page);
        }
        if (frameCommonMethods == null) {
            frameCommonMethods = new FrameCommonMethods(new_page);
        }
    }

    /**
     * Step to open the popup and initialize new_page + helpers.
     * Call this step before using any steps that act on 'new_page'.
     */
    @When("^we open the new page by clicking Open URL$")
    public void openNewPageByClickingOpenUrl() {
        // Wait for popup that opens after clicking the button on the main page
        new_page = page.waitForPopup(() -> {
            page.getByRole(
                    AriaRole.BUTTON,
                    new Page.GetByRoleOptions().setName("Open URL")
            ).click();
        });

        // Initialize helpers now that we have the popup page
        pageCommonMethods = new PageCommonMethods(new_page);
        frameCommonMethods = new FrameCommonMethods(new_page);

        System.out.println("[NewPageCommonSteps] New popup page opened: " + new_page.url());
    }

    /**
     * Your original helper method – unchanged.
     */
    public String getConfirmationId(String locatorValue) {
        // Extract the actual locator by splitting the locatorValue at the first colon
        int firstColonIndex = locatorValue.indexOf(":");
        if (firstColonIndex != -1 && firstColonIndex + 1 < locatorValue.length()) {
            return locatorValue.substring(firstColonIndex + 1); // Get substring after the first colon
        }
        return ""; // Return an empty string if no colon or nothing follows the first colon
    }

    @Then("^I call dummy step$")
    public void icallDummyStep() {
        System.out.println("Dummy step");
    }

    /**
     * Click on an element in the popup page using your PageCommonMethods.
     * Make sure the popup is opened first by calling the "open new page" step.
     */
    @Then("^we click on new page (.*?) locator (.*?)$")
    public void weClickActionNewOnPage(String element, String locator) {
        ensurePopupInitialized();
        pageCommonMethods.click(new_page, element, locator);
    }

    // If you need any frame-related steps later, they can also call ensurePopupInitialized()
    // and use frameCommonMethods with plad_frame / atomic_frame / pop_frame, etc.
}
