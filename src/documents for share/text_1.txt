import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

@When("we click all available documents on doc frame")
public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
    int total = ReturnTotalNumberOfElement();
    String basePath = ConfigurationProperties.getValue("downloadDocument"); // e.g. C:/Downloads/document
    String ext = ".pdf";
    String screenshotPath = "test-output/screenshots/body.png";

    if (total <= 0) {
        weCaptureScreenshotOnDirectorPage("director_page", "body", screenshotPath);
        weReportStringValue("[INFO]", "No documents available to download.");
        return;
    }

    for (int i = 1; i <= total; i++) {
        String key = toDocumentKey(i); // document_one, document_two, ...
        String targetFile = String.format("%s_%02d%s", basePath, i, ext);

        boolean success = downloadWithRetryAndVerify(
                "director_page",
                key,
                targetFile,
                screenshotPath,
                3,       // max attempts
                1200     // wait between attempts (ms)
        );

        if (success) {
            System.out.printf("[INFO] Downloaded %s using key %s%n", targetFile, key);
        } else {
            System.out.printf("[WARN] Giving up after retries: %s (key=%s)%n", targetFile, key);
        }

        Thread.sleep(300); // small pause between documents
    }
}

/**
 * Attempts download several times.
 * After each attempt, verifies file exists and has non-zero size.
 * On failure: screenshot + continue retries; never throws out to break the loop.
 */
private boolean downloadWithRetryAndVerify(
        String pageName,
        String docKey,
        String targetFile,
        String screenshotPath,
        int maxAttempts,
        long waitBetweenAttemptsMs
) {
    for (int attempt = 1; attempt <= maxAttempts; attempt++) {
        try {
            // Clean old leftover file so verification is real
            safeDeleteIfExists(targetFile);

            // Your current flow
            weDoubleClickActionOnDirectorDocFrame(pageName, docKey);
            weDownloadOnDirectorDocFrame(pageName, docKey, targetFile);

            // Verify download really happened
            if (waitForDownloadedFile(targetFile, 8000, 250)) { // timeout 8s, poll 250ms
                return true;
            }

            // If not downloaded, treat as failure and retry
            weCaptureScreenshotOnDirectorPage(pageName, "body", screenshotPath);
            System.out.printf("[WARN] Attempt %d/%d: file not found or empty: %s (key=%s)%n",
                    attempt, maxAttempts, targetFile, docKey);

        } catch (Exception e) {
            // Always screenshot and continue to next attempt
            try {
                weCaptureScreenshotOnDirectorPage(pageName, "body", screenshotPath);
            } catch (Exception ignored) {}

            System.out.printf("[WARN] Attempt %d/%d failed for %s (key=%s): %s%n",
                    attempt, maxAttempts, targetFile, docKey, e.getMessage());
        }

        // Small backoff before retry
        try {
            Thread.sleep(waitBetweenAttemptsMs);
        } catch (InterruptedException ie) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
    return false;
}

private boolean waitForDownloadedFile(String targetFile, long timeoutMs, long pollMs) {
    long end = System.currentTimeMillis() + timeoutMs;
    Path p = Paths.get(targetFile);

    while (System.currentTimeMillis() < end) {
        try {
            if (Files.exists(p)) {
                long size = Files.size(p);
                if (size > 0) return true; // downloaded and not empty
            }
        } catch (Exception ignored) {}

        try {
            Thread.sleep(pollMs);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return false;
        }
    }
    return false;
}

private void safeDeleteIfExists(String targetFile) {
    try {
        Path p = Paths.get(targetFile);
        if (Files.exists(p)) Files.delete(p);
    } catch (Exception ignored) {}
}

private String toDocumentKey(int i) {
    switch (i) {
        case 1: return "document_one";
        case 2: return "document_two";
        case 3: return "document_three";
        case 4: return "document_four";
        case 5: return "document_five";
        case 6: return "document_six";
        case 7: return "document_seven";
        case 8: return "document_eight";
        case 9: return "document_nine";
        case 10:return "document_ten";
        default: throw new IllegalArgumentException("Unsupported document index: " + i);
    }
}
