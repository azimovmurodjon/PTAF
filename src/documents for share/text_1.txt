 @When("we click all available documents on doc frame")
    public void weClickAllAvailableDocumentsOnDocFrame() throws InterruptedException {
        int total = ReturnTotalNumberOfElement();
        String basePath = ConfigurationProperties.getValue("downloadDocument"); // e.g. C:/Downloads/document
        String ext = ".pdf";
        String filePath = "test-output/screenshots/body.png";
        if (total <= 0) {
            weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
            weReportStringValue("[INFO]", "No documents available to download.");
            return;
        }

        for (int i = 1; i <= total; i++) {
            String key = toDocumentKey(i); // document_one, document_two, ...
            String targetFile = String.format("%s_%02d%s", basePath, i, ext); // unique name

            try {
                weDoubleClickActionOnDirectorDocFrame("director_page", key);
//                weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);

                if (weVerifyOnDirectorMarkupFrameIsBooleanVisible("director_page", "deposit_main_title"))
                    weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                else {
                    try {
                        System.out.println("[INFO] Try to Download");
                        weDownloadOnDirectorDocFrame("director_page", key, targetFile);
                        weCaptureScreenshotOnDirectorPage("director_page", "body", filePath);
                        System.out.printf("[INFO] Downloaded %s using key %s%n", targetFile, key);
                        Thread.sleep(500); // small pause; increase only if needed

                    } catch (Exception e) {
                        System.out.printf("[WARN] Failed to download %s (%s): %s%n", targetFile, key, e.getMessage());
                    }
                }
            } catch (Exception e) {

                System.out.println("[WARN] Failed to capture screenshot");
            }
        }
    }
